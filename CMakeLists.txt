function(sc_configure_core target)
  set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
  target_include_directories(${target} PUBLIC src)
  if(NOT MSVC)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(${target} PUBLIC m)
  endif()
  if(SC_COVERAGE)
    if(EMSCRIPTEN OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
      message(FATAL_ERROR "SC_COVERAGE requires native GCC and its matching gcov")
    endif()
    target_compile_options(${target} PUBLIC --coverage -O0 -g)
    target_link_options(${target} PUBLIC --coverage)
  endif()
  if(SC_SANITIZE AND NOT MSVC AND NOT EMSCRIPTEN)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endfunction()
sc_configure_core(sc_core)
