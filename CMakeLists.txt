cmake_minimum_required(VERSION 3.22)
project(starship_crawl VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(EMSCRIPTEN)
  # Emscripten's EM_ASM bridge (also used by SDL) requires GNU C extensions.
  set(CMAKE_C_EXTENSIONS ON)
endif()
option(SC_BUILD_GAME "Build the SDL/OpenGL application" ON)
option(SC_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(SC_COVERAGE "Instrument native GCC unit-test coverage" OFF)

set(SC_CORE_SOURCES src/content.c src/generation.c src/history.c src/interior.c src/world.c src/progression.c src/story.c src/expedition.c src/game.c src/save.c src/shelter.c src/navigation.c src/psionics.c src/equipment.c)
add_library(sc_core ${SC_CORE_SOURCES})
set_target_properties(sc_core PROPERTIES C_EXTENSIONS OFF)
target_include_directories(sc_core PUBLIC src)
if(NOT MSVC)
  target_compile_options(sc_core PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(sc_core PUBLIC m)
endif()
if(SC_COVERAGE)
  if(EMSCRIPTEN OR NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "SC_COVERAGE requires native GCC and its matching gcov")
  endif()
  target_compile_options(sc_core PUBLIC --coverage -O0 -g)
  target_link_options(sc_core PUBLIC --coverage)
endif()
if(SC_SANITIZE AND NOT MSVC AND NOT EMSCRIPTEN)
  target_compile_options(sc_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sc_core PUBLIC -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING AND NOT EMSCRIPTEN)
  add_executable(sc_tests tests/test_core.c)
  target_link_libraries(sc_tests PRIVATE sc_core)
  target_compile_definitions(sc_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME core COMMAND sc_tests)
  set_tests_properties(core PROPERTIES LABELS "unit" TIMEOUT 60)
  add_executable(sc_interior_tests tests/test_interiors.c)
  target_link_libraries(sc_interior_tests PRIVATE sc_core)
  target_compile_definitions(sc_interior_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME generated_interiors COMMAND sc_interior_tests)
  set_tests_properties(generated_interiors PROPERTIES LABELS "unit" TIMEOUT 60)

  # Compile the same core with test-only allocator/I/O wrappers. Production
  # builds do not contain fault injection or a test control API.
  add_library(sc_core_fault STATIC ${SC_CORE_SOURCES})
  target_include_directories(sc_core_fault PUBLIC src)
  if(MSVC)
    target_compile_options(sc_core_fault PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/tests/fault_alloc.h")
  else()
    target_compile_options(sc_core_fault PRIVATE -Wall -Wextra -Wpedantic -include "${CMAKE_CURRENT_SOURCE_DIR}/tests/fault_alloc.h")
    target_link_libraries(sc_core_fault PUBLIC m)
  endif()
  if(SC_SANITIZE AND NOT MSVC)
    target_compile_options(sc_core_fault PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(sc_core_fault PUBLIC -fsanitize=address,undefined)
  endif()
  if(SC_COVERAGE)
    target_compile_options(sc_core_fault PUBLIC --coverage -O0 -g)
    target_link_options(sc_core_fault PUBLIC --coverage)
  endif()
  add_executable(sc_edge_tests tests/test_edges.c tests/fault_alloc.c)
  target_link_libraries(sc_edge_tests PRIVATE sc_core_fault)
  target_compile_definitions(sc_edge_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME edge_cases COMMAND sc_edge_tests)
  set_tests_properties(edge_cases PROPERTIES LABELS "unit" TIMEOUT 60)
  add_executable(sc_navigation_tests tests/test_navigation_psionics.c)
  target_link_libraries(sc_navigation_tests PRIVATE sc_core)
  target_compile_definitions(sc_navigation_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME navigation_psionics COMMAND sc_navigation_tests)
  set_tests_properties(navigation_psionics PROPERTIES LABELS "unit" TIMEOUT 60)
  add_executable(sc_shelter_tests tests/test_shelter.c)
  target_link_libraries(sc_shelter_tests PRIVATE sc_core)
  target_compile_definitions(sc_shelter_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME shelter COMMAND sc_shelter_tests)
  set_tests_properties(shelter PROPERTIES LABELS "unit" TIMEOUT 60)
  add_test(NAME shelter_journey COMMAND sc_shelter_tests --journey)
  set_tests_properties(shelter_journey PROPERTIES LABELS "gameplay" TIMEOUT 60)
  add_executable(sc_equipment_tests tests/test_equipment.c)
  target_link_libraries(sc_equipment_tests PRIVATE sc_core)
  target_compile_definitions(sc_equipment_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME equipment COMMAND sc_equipment_tests)
  set_tests_properties(equipment PROPERTIES LABELS "unit" TIMEOUT 60)
  add_test(NAME equipment_journey COMMAND sc_equipment_tests --journey)
  set_tests_properties(equipment_journey PROPERTIES LABELS "gameplay" TIMEOUT 60)
  add_executable(sc_campaign_tests tests/test_campaign.c)
  target_link_libraries(sc_campaign_tests PRIVATE sc_core)
  add_test(NAME solo_campaign COMMAND sc_campaign_tests)
  set_tests_properties(solo_campaign PROPERTIES LABELS "gameplay" TIMEOUT 60)
  add_test(NAME generated_expeditions COMMAND sc_campaign_tests --story)
  set_tests_properties(generated_expeditions PROPERTIES LABELS "gameplay" TIMEOUT 180)
  add_test(NAME history_expeditions COMMAND sc_campaign_tests --history)
  set_tests_properties(history_expeditions PROPERTIES LABELS "gameplay" TIMEOUT 240)
  add_test(NAME continuing_expeditions COMMAND sc_campaign_tests --saga)
  set_tests_properties(continuing_expeditions PROPERTIES LABELS "gameplay" TIMEOUT 600)
  add_executable(sc_expedition_tests tests/test_expedition.c)
  target_link_libraries(sc_expedition_tests PRIVATE sc_core)
  add_test(NAME reactive_aftermath COMMAND sc_expedition_tests)
  set_tests_properties(reactive_aftermath PROPERTIES LABELS "unit" TIMEOUT 120)
  add_executable(sc_story_tests tests/test_story.c)
  target_link_libraries(sc_story_tests PRIVATE sc_core)
  target_compile_definitions(sc_story_tests PRIVATE SC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME narrative COMMAND sc_story_tests)
  set_tests_properties(narrative PROPERTIES LABELS "unit" TIMEOUT 60)
endif()

if(SC_BUILD_GAME)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3
    URL https://codeload.github.com/libsdl-org/SDL/tar.gz/f5e5f6588921eed3d7d048ce43d9eb1ff0da0ffc
    URL_HASH SHA256=7f80bce51b7179bab472e75b90ce8ac06b162b3da8e19feadecee5acea07dbfb
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
  add_executable(starship_crawl src/main.c src/render.c src/gl_api.c)
  target_link_libraries(starship_crawl PRIVATE sc_core SDL3::SDL3)
  if(NOT MSVC)
    target_compile_options(starship_crawl PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  if(EMSCRIPTEN)
    set_target_properties(starship_crawl PROPERTIES SUFFIX ".html")
    target_link_options(starship_crawl PRIVATE
      "-sMIN_WEBGL_VERSION=2" "-sMAX_WEBGL_VERSION=2"
      "-sALLOW_MEMORY_GROWTH=1" "-sINITIAL_MEMORY=67108864" "-sMAXIMUM_MEMORY=268435456"
      "-sMINIFY_HTML=0"
      "-sEXPORTED_FUNCTIONS=['_main','_sc_web_start','_sc_web_capture','_sc_web_look','_sc_web_save','_sc_web_state','_sc_web_action','_sc_web_import','_sc_web_fingerprint']"
      "-sEXPORTED_RUNTIME_METHODS=['ccall','FS']" "-sFORCE_FILESYSTEM=1"
      "-lidbfs.js" "--shell-file=${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html"
      "--pre-js=${CMAKE_CURRENT_SOURCE_DIR}/web/storage.js")
    set_property(TARGET starship_crawl APPEND PROPERTY LINK_DEPENDS
      ${CMAKE_CURRENT_SOURCE_DIR}/web/shell.html ${CMAKE_CURRENT_SOURCE_DIR}/web/storage.js)
  endif()
endif()
